同一块“布”的六种视图

布面由顶点着色器按正弦波起伏；片元着色器计算光照、采样纹理。调试视图把中间结果直接显示成颜色，是检查着色器最常用的方法。

最终效果UV法线光照纹理高度

着色器代码（GLSL）

// 顶点着色器：每个顶点运行一次，改变位置并算出法线
uniform float uTime, uAmp, uFreq;
varying vec2 vUv;  varying vec3 vNormal;  varying float vHeight;
void main() {
  vec3 p = position;
  float phase = p.x * uFreq + uTime;
  p.z += uAmp * sin(phase);                  // 顶点位移
  float slope = uAmp * uFreq * cos(phase);   // 斜率 dz/dx
  vNormal = normalize(normalMatrix * vec3(-slope, 0.0, 1.0));
  vUv = uv;  vHeight = p.z;
  gl_Position = projectionMatrix * modelViewMatrix * vec4(p, 1.0);
}

// 片元着色器：每个像素运行一次，算出颜色
uniform sampler2D uMap;  uniform vec3 uLightDir;  uniform int uView;
void main() {
  vec3 n = normalize(vNormal);   // 插值后长度会变，重新归一化
  float light = 0.25 + 0.75 * max(dot(n, uLightDir), 0.0);
  vec3 tex = texture2D(uMap, vUv * 2.0).rgb;   // 采样，重复两次
  vec3 color = tex * light;                    // 最终效果
  if (uView == 1) color = vec3(vUv, 0.0);      // UV
  if (uView == 2) color = n * 0.5 + 0.5;       // 法线：−1…1 → 0…1
  if (uView == 3) color = vec3(light);         // 光照
  if (uView == 4) color = tex;                 // 纹理
  if (uView == 5) color = vec3(vHeight / uAmp * 0.5 + 0.5);  // 高度
  gl_FragColor = vec4(color, 1.0);
  // 颜色从线性空间转换成屏幕的 sRGB；数据视图直接显示数值
  if (uView == 0 || uView == 4) gl_FragColor = linearToOutputTexel(gl_FragColor);
}
