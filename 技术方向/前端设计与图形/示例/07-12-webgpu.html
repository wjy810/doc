WebGL 2 ＋ GLSL

各主流浏览器都支持。

uniform float uTime, uAspect;
in vec2 vUv;  out vec4 outColor;
void main() {
  vec2 p = (vUv - 0.5) * vec2(uAspect, 1.0);
  float r = length(p);
  float wave = 0.5 + 0.5 * sin(r * 40.0 - uTime * 3.0);
  vec3 col = mix(vec3(0.12, 0.23, 0.43), vec3(0.96, 0.78, 0.55),
                 wave * smoothstep(0.7, 0.0, r));
  outColor = vec4(col, 1.0);
}

struct U { time: f32, aspect: f32 };
@group(0) @binding(0) var<uniform> u: U;
@fragment
fn fs(@location(0) uv: vec2f) -> @location(0) vec4f {
  let p = (uv - 0.5) * vec2f(u.aspect, 1.0);
  let r = length(p);
  let wave = 0.5 + 0.5 * sin(r * 40.0 - u.time * 3.0);
  let col = mix(vec3f(0.12, 0.23, 0.43), vec3f(0.96, 0.78, 0.55),
                wave * smoothstep(0.7, 0.0, r));
  return vec4f(col, 1.0);
}

three.js 节点材质（TSL）

用 JavaScript 组合节点，框架按后端生成 WGSL 或 GLSL。

const p = uv().sub(0.5).mul(vec2(aspect, 1));
const r = length(p);
const wave = sin(r.mul(40).sub(time.mul(3))).mul(0.5).add(0.5);
material.colorNode = mix(vec3(0.12, 0.23, 0.43), vec3(0.96, 0.78, 0.55),
                         wave.mul(smoothstep(0.7, 0, r)));
